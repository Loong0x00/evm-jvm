// Test-only generated streams: complete code-length tree for lengths0 and1;
// singleton EOB literal tree; no matches and empty distance alphabet.
func countStream(nlit, ndist int) []byte {
	w := bitWriter{}
	w.bits(1, 1)
	w.bits(2, 2)
	w.bits(uint(nlit-257), 5)
	w.bits(uint(ndist-1), 5)
	w.bits(14, 4)
	order := []int{16, 17, 18, 0, 8, 7, 9, 6, 10, 5, 11, 4, 12, 3, 13, 2, 14, 1}
	for _, s := range order {
		if s == 0 || s == 1 {
			w.bits(1, 3)
		} else {
			w.bits(0, 3)
		}
	}
	for i := 0; i < nlit+ndist; i++ {
		if i == 256 {
			w.code(1, 1)
		} else {
			w.code(0, 1)
		}
	}
	w.code(0, 1)
	return w.b
}

func writeCheckOnly(path string, b []byte, _ os.FileMode) error {
	old, e := os.ReadFile(path)
	if e != nil {
		return e
	}
	if !bytes.Equal(old, b) {
		return fmt.Errorf("fixture bytes changed: %s", path)
	}
	return nil
}
