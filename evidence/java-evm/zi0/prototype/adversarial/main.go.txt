const ordinary = uint64(100_000_000)

type artifact struct {
	ABI      json.RawMessage `json:"abi"`
	Bytecode string          `json:"bytecode"`
	Runtime  string          `json:"runtime"`
}
type ent struct {
	name       string
	data       []byte
	method     uint16
	descriptor int
	flags      uint16
	extra      []byte
	comment    []byte
}
type built struct {
	raw           []byte
	cd, eocd      uint32
	locals, datas []uint32
	central       [][]byte
}
type report struct {
	Name          string `json:"name"`
	Size          int    `json:"size"`
	Pages         int    `json:"pages"`
	Want          string `json:"want"`
	Got           string `json:"got"`
	AS0Create     uint64 `json:"as0_create_gas"`
	AS0Append     uint64 `json:"as0_append_gas"`
	AS0AppendMax  uint64 `json:"as0_append_max_gas"`
	AS0Seal       uint64 `json:"as0_seal_gas"`
	ScannerCreate uint64 `json:"scanner_create_gas"`
	Scan          uint64 `json:"scan_gas"`
	Count         uint32 `json:"count,omitempty"`
	CDSize        uint32 `json:"cd_size,omitempty"`
	Root          string `json:"index_root,omitempty"`
}

func put16(b *bytes.Buffer, v uint16) { binary.Write(b, binary.LittleEndian, v) }
func put32(b *bytes.Buffer, v uint32) { binary.Write(b, binary.LittleEndian, v) }
func deflate(x []byte) []byte {
	var b bytes.Buffer
	w, _ := flate.NewWriter(&b, flate.DefaultCompression)
	w.Write(x)
	w.Close()
	return b.Bytes()
}
func makezip(es []ent, prefix int, order []int, comment []byte) built {
	var out bytes.Buffer
	out.Write(bytes.Repeat([]byte{'X'}, prefix))
	z := built{}
	for _, e := range es {
		z.locals = append(z.locals, uint32(out.Len()))
		name := []byte(e.name)
		payload := e.data
		if e.method == 8 {
			payload = deflate(e.data)
		}
		flags := uint16(0x800) | e.flags
		if e.descriptor >= 0 {
			flags |= 8
		}
		crc := crc32.ChecksumIEEE(e.data)
		put32(&out, 0x04034b50)
		put16(&out, 20)
		put16(&out, flags)
		put16(&out, e.method)
		put16(&out, 0)
		put16(&out, 0)
		if e.descriptor >= 0 {
			put32(&out, 0)
			put32(&out, 0)
			put32(&out, 0)
		} else {
			put32(&out, crc)
			put32(&out, uint32(len(payload)))
			put32(&out, uint32(len(e.data)))
		}
		put16(&out, uint16(len(name)))
		put16(&out, uint16(len(e.extra)))
		out.Write(name)
		out.Write(e.extra)
		z.datas = append(z.datas, uint32(out.Len()))
		out.Write(payload)
		if e.descriptor >= 0 {
			if e.descriptor == 1 {
				put32(&out, 0x08074b50)
			}
			put32(&out, crc)
			put32(&out, uint32(len(payload)))
			put32(&out, uint32(len(e.data)))
		}
		var c bytes.Buffer
		put32(&c, 0x02014b50)
		put16(&c, 0x0314)
		put16(&c, 20)
		put16(&c, flags)
		put16(&c, e.method)
		put16(&c, 0)
		put16(&c, 0)
		put32(&c, crc)
		put32(&c, uint32(len(payload)))
		put32(&c, uint32(len(e.data)))
		put16(&c, uint16(len(name)))
		put16(&c, uint16(len(e.extra)))
		put16(&c, uint16(len(e.comment)))
		put16(&c, 0)
		put16(&c, 0)
		put32(&c, 0)
		put32(&c, z.locals[len(z.locals)-1])
		c.Write(name)
		c.Write(e.extra)
		c.Write(e.comment)
		z.central = append(z.central, c.Bytes())
	}
	z.cd = uint32(out.Len())
	if order == nil {
		order = make([]int, len(es))
		for i := range order {
			order[i] = i
		}
	}
	for _, i := range order {
		out.Write(z.central[i])
	}
	cds := uint32(out.Len()) - z.cd
	z.eocd = uint32(out.Len())
	put32(&out, 0x06054b50)
	put16(&out, 0)
	put16(&out, 0)
	put16(&out, uint16(len(es)))
	put16(&out, uint16(len(es)))
	put32(&out, cds)
	put32(&out, z.cd)
	put16(&out, uint16(len(comment)))
	out.Write(comment)
	z.raw = out.Bytes()
	return z
}
func cfg(st *state.StateDB, owner common.Address, gas uint64) *runtime.Config {
	zero := common.Hash{}
	return &runtime.Config{ChainConfig: params.AllDevChainProtocolChanges, Difficulty: new(big.Int), Origin: owner, Coinbase: common.Address{}, BlockNumber: big.NewInt(1), Time: 1, GasLimit: gas, GasPrice: new(big.Int), Value: new(big.Int), BaseFee: new(big.Int).SetUint64(params.InitialBaseFee), BlobBaseFee: new(big.Int).SetUint64(params.BlobTxMinBlobGasprice), Random: &zero, State: st, GetHashFn: func(uint64) common.Hash { return common.Hash{} }, EVMConfig: vm.Config{}}
}
func enc(typesx []string, v ...interface{}) []byte {
	var a abi.Arguments
	for _, s := range typesx {
		t, _ := abi.NewType(s, "", nil)
		a = append(a, abi.Argument{Type: t})
	}
	x, e := a.Pack(v...)
	if e != nil {
		panic(e)
	}
	return x
}
func pi(at common.Address, i uint32, raw []byte) common.Hash {
	return crypto.Keccak256Hash(enc([]string{"bytes32", "address", "uint32", "uint32", "bytes32"}, crypto.Keccak256Hash([]byte("MC-AS0/page/v1")), at, i, uint32(len(raw)), crypto.Keccak256Hash(raw)))
}
func call(st *state.StateDB, owner, to common.Address, data []byte, gas uint64) ([]byte, uint64, error) {
	o, left, e := runtime.Call(to, data, cfg(st, owner, gas))
	return o, gas - left, e
}
func getter(st *state.StateDB, owner, to common.Address, a abi.ABI, name string) []interface{} {
	d, _ := a.Pack(name)
	o, _, e := call(st, owner, to, d, ordinary)
	if e != nil {
		panic(e)
	}
	v, e := a.Unpack(name, o)
	if e != nil {
		panic(e)
	}
	return v
}
func badcode(out []byte) string {
	if len(out) != 36 {
		return "revert"
	}
	return fmt.Sprintf("selector0x%x(%d)", out[:4], new(big.Int).SetBytes(out[4:]).Uint64())
}
func one(name string, z built, wantOK bool, mut func([]byte), proto artifact, pa abi.ABI) report {
	raw := append([]byte(nil), z.raw...)
	if name == "truncated" {
		raw = raw[:len(raw)-1]
	}
	if mut != nil {
		mut(raw)
	}
	r := report{Name: name, Size: len(raw), Pages: (len(raw) + 8191) / 8192}
	owner := common.HexToAddress("0x5a49300000000000000000000000000000000001")
	db := rawdb.NewMemoryDatabase()
	td := triedb.NewDatabase(db, triedb.HashDefaults)
	defer td.Close()
	st, e := state.New(types.EmptyRootHash, state.NewDatabase(td, nil))
	if e != nil {
		panic(e)
	}
	init, _ := as0.InitCode()
	args, _ := pa.Pack("", uint32(len(raw)))
	code := append(init, args...)
	_, store, left, e := runtime.Create(code, cfg(st, owner, ordinary))
	r.AS0Create = ordinary - left
	if e != nil {
		panic(e)
	}
	var cursor uint32
	for i := uint32(0); int(cursor) < len(raw); i++ {
		n := 8192
		if len(raw)-int(cursor) < n {
			n = len(raw) - int(cursor)
		}
		page := raw[cursor : uint32(cursor)+uint32(n)]
		d, _ := pa.Pack("append", cursor, page, pi(store, i, page))
		_, g, e := call(st, owner, store, d, ordinary)
		if e != nil {
			panic(fmt.Sprintf("append %s: %v", name, e))
		}
		r.AS0Append += g
		if g > r.AS0AppendMax {
			r.AS0AppendMax = g
		}
		cursor += uint32(n)
	}
	chain := getter(st, owner, store, pa, "chainCommitment")[0].([32]byte)
	var root [32]byte
	if name == "unsealed-store" {
		root[31] = 1
	} else {
		d, _ := pa.Pack("seal", cursor, chain)
		_, r.AS0Seal, e = call(st, owner, store, d, ordinary)
		if e != nil {
			panic(e)
		}
		root = getter(st, owner, store, pa, "contentCommitment")[0].([32]byte)
	}
	pb, _ := hex.DecodeString(proto.Bytecode[2:])
	_, scanner, left, e := runtime.Create(pb, cfg(st, owner, ordinary))
	r.ScannerCreate = ordinary - left
	if e != nil {
		panic(e)
	}
	sa, e := abi.JSON(bytes.NewReader(proto.ABI))
	if e != nil {
		panic(e)
	}
	callStore := store
	callRoot := root
	if name == "wrong-content-root" {
		callRoot[0] ^= 1
	}
	if name == "wrong-code-binding" {
		callStore = common.HexToAddress("0xdead")
	}
	d, _ := sa.Pack("scan", callStore, callRoot)
	var out []byte
	out, r.Scan, e = call(st, owner, scanner, d, ordinary)
	if e != nil {
		r.Got = badcode(out)
	} else {
		v, u := sa.Unpack("scan", out)
		if u != nil {
			panic(u)
		}
		r.Got = "ok"
		r.Root = common.Hash(v[0].([32]byte)).Hex()
		r.Count = v[1].(uint32)
		r.CDSize = v[3].(uint32)
	}
	if wantOK {
		r.Want = "ok"
	} else {
		r.Want = "reject"
	}
	if (e == nil) != wantOK {
		fmt.Fprintf(os.Stderr, "COUNTEREXAMPLE %s wantOK=%v got=%s\n", name, wantOK, r.Got)
	}
	return r
}
func main() {
	start := time.Now()
	doc, err := os.ReadFile("build/java-diff-mc-zi0-parent/original/prototype.json")
	if err != nil {
		panic(err)
	}
	var proto artifact
	if json.Unmarshal(doc, &proto) != nil {
		panic("json")
	}
	pa, err := as0.ABI()
	if err != nil {
		panic(err)
	}
	cases := []report{}
	add := func(n string, z built, ok bool, m func([]byte)) { cases = append(cases, one(n, z, ok, m, proto, pa)) }
	for _, bit := range []uint16{4, 7, 8, 9, 10, 12, 13, 14, 15} {
		add(fmt.Sprintf("reserved-flag-bit-%d", bit), makezip([]ent{{"x", []byte("payload"), 8, -1, 1 << bit, nil, nil}}, 0, nil, nil), false, nil)
	}
	for _, bit := range []uint16{1, 2} {
		add(fmt.Sprintf("stored-deflate-option-bit-%d", bit), makezip([]ent{{"x", []byte("p"), 0, -1, 1 << bit, nil, nil}}, 0, nil, nil), false, nil)
	}
	loc := makezip(nil, 20, nil, nil)
	binary.LittleEndian.PutUint32(loc.raw, 0x07064b50)
	binary.LittleEndian.PutUint32(loc.raw[16:], 1)
	add("zip64-locator-before-empty-eocd", loc, false, nil)
	// Treat CEN's first12 bytes as the missing unsigned data descriptor.
	// A metadata-only parser need not inflate these intentionally opaque bytes,
	// but it must reject a local record ending INSIDE its own central directory.
	const comp = 0x00140314
	z := makezip([]ent{{"x", make([]byte, comp), 0, 0, 0, nil, nil}}, 0, nil, nil)
	z.raw = append(append([]byte(nil), z.raw[:z.cd-12]...), z.raw[z.cd:]...)
	z.cd -= 12
	z.eocd -= 12
	binary.LittleEndian.PutUint16(z.raw[8:], 8)
	binary.LittleEndian.PutUint16(z.raw[z.cd+10:], 8)
	binary.LittleEndian.PutUint32(z.raw[z.cd+16:], 0x02014b50)
	binary.LittleEndian.PutUint32(z.raw[z.cd+24:], 0x00080808)
	binary.LittleEndian.PutUint32(z.raw[z.eocd+16:], z.cd)
	add("last-descriptor-overlaps-directory", z, false, nil)
	// Same tail length as the measured control, but filled with signature prefixes.
	add("dense-eocd-signatures", makezip(nil, 70000, nil, bytes.Repeat([]byte("PK\x05\x06"), 16383)), true, nil)
	sort.Slice(cases, func(i, j int) bool { return cases[i].Name < cases[j].Name })
	var usage syscall.Rusage
	_ = syscall.Getrusage(syscall.RUSAGE_SELF, &usage)
	sum := map[string]interface{}{"elapsed_ms": time.Since(start).Milliseconds(), "max_rss_kib": usage.Maxrss, "prototype_init_bytes": (len(proto.Bytecode) - 2) / 2, "prototype_runtime_bytes": (len(proto.Runtime) - 2) / 2, "prototype_runtime_keccak": crypto.Keccak256Hash(common.FromHex(proto.Runtime)).Hex(), "cases": cases}
	encj, _ := json.MarshalIndent(sum, "", "  ")
	fmt.Println(string(encj))
	for _, r := range cases {
		if (r.Got == "ok") != (r.Want == "ok") {
			os.Exit(1)
		}
	}
}
