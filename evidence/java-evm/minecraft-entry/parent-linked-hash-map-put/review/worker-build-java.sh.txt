#!/usr/bin/env bash
set -euo pipefail
cd "$(dirname "$0")"
mkdir -p build/context/java/util build/classes
cp bootstrap/java/util/*.java build/context/java/util/
# Compile context only: P58's compact Map source omits Java 8 default-method
# declarations needed by unchanged HashMap.java's @Override annotations. Add
# these declarations only to the disposable compiler context; do not embed its
# Map/HashMap/other old class outputs. No access/finality semantics are changed.
python3 - <<'PY'
p='build/context/java/util/Map.java'
s=open(p).read()
needle='    Set<Map.Entry<K,V>> entrySet();\n'
assert s.count(needle) == 1
s=s.replace(needle, needle + '''    default V getOrDefault(Object key, V defaultValue) { return null; }
    default void forEach(java.util.function.BiConsumer<? super K, ? super V> action) { }
    default void replaceAll(java.util.function.BiFunction<? super K, ? super V, ? extends V> function) { }
    default V putIfAbsent(K key, V value) { return null; }
    default boolean remove(Object key, Object value) { return false; }
    default boolean replace(K key, V oldValue, V newValue) { return false; }
    default V replace(K key, V value) { return null; }
    default V computeIfAbsent(K key, java.util.function.Function<? super K, ? extends V> function) { return null; }
    default V computeIfPresent(K key, java.util.function.BiFunction<? super K, ? super V, ? extends V> function) { return null; }
    default V compute(K key, java.util.function.BiFunction<? super K, ? super V, ? extends V> function) { return null; }
    default V merge(K key, V value, java.util.function.BiFunction<? super V, ? super V, ? extends V> function) { return null; }
''')
open(p,'w').write(s)
PY
javac --release 8 -g:none -Xlint:-options -d build/classes \
  build/context/java/util/AbstractCollection.java \
  build/context/java/util/AbstractMap.java \
  build/context/java/util/AbstractSet.java \
  build/context/java/util/Collection.java \
  build/context/java/util/HashMap.java \
  build/context/java/util/Iterator.java \
  build/context/java/util/LinkedHashMap.java \
  build/context/java/util/Map.java \
  build/context/java/util/Set.java \
  bootstrap/java/util/LinkedHashMapEntry.java \
  bootstrap/java/util/LinkedHashMapEntrySet.java \
  bootstrap/java/util/LinkedHashMapEntryIterator.java
python3 - <<'PY'
from pathlib import Path
for name in ('LinkedHashMap','LinkedHashMapEntry','LinkedHashMapEntrySet','LinkedHashMapEntryIterator'):
    data=Path('build/classes/java/util/'+name+'.class').read_bytes()
    Path('bootstrap/java/util/'+name+'.class.hex').write_text(data.hex()+'\n')
PY
sha256sum bootstrap/java/util/LinkedHashMap.class.hex bootstrap/java/util/LinkedHashMapEntry.class.hex bootstrap/java/util/LinkedHashMapEntrySet.class.hex bootstrap/java/util/LinkedHashMapEntryIterator.class.hex
